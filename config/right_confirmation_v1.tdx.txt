{右侧量价确认V1：条件选股；日线；使用前复权；EXTDATA6/7必须更新}
{研究候选，不表示未来主升概率；参数尚未样本外验证}
R120:=EXTDATA_USER(6,0)/10;
R250:=EXTDATA_USER(7,0)/10;
M5:=MA(C,5);
M10:=MA(C,10);
M20:=MA(C,20);
M60:=MA(C,60);
G1:=R120+R250>180;
G2:=C>M20 AND COUNT(C>MA(C,250),30)>=20
 AND COUNT(C>MA(C,200),30)>=20
 AND (COUNT(C>M20,10)>=7 OR (COUNT(C>M10,4)>=3 AND COUNT(C>M20,4)>=3));
HA:=HHVBARS(H,20);
LA:=IF(HA=0,0,LLVBARS(L,MAX(HA,1)));
DD:=(REF(H,HA)-REF(L,LA))/REF(H,HA);
{仅修正COUNT(X,0)历史累计陷阱；与原公式分开保存}
G3:=DD<=0.25 AND IF(HA=0,1,COUNT(DD>0.25,MAX(HA,1))=0)
 AND C/HHV(C,120)>0.8;
G4:=(EVERY(M20>=REF(M20,1),5) AND EVERY(M10>=M20,5))
 OR (M10>=REF(M10,1) AND M20>=REF(M20,1) AND M10>=M20);
G5:=CAPITAL>0 AND VOL/CAPITAL*100<10;
G6:=COUNT(C>M5,10)>=6;
G7:=COUNT(AMOUNT>100000000,10)>=7;
G8:=COUNT(C<M5,3)<=1;
BASE:=G1 AND G2 AND G3 AND G4 AND G5 AND G6 AND G7 AND G8;
TREND:=C>M20 AND M20>M60 AND M20>REF(M20,5) AND M60>REF(M60,5);
BIAS20:=C/M20-1;
P20:=REF(HHV(H,20),1);
V20:=REF(MA(VOL,20),1);
VR:=IF(V20>0,VOL/V20,0);
CLV:=IF(H>L,(C-L)/(H-L),0);
WIDTH10:=REF(HHV(H,10),1)/REF(LLV(L,10),1)-1;
DRY:=IF(V20>0,REF(MA(VOL,5),1)/V20,999);
{突破前的平台窄、量能收缩；突破当日恢复成交并收于上部}
BR:=C>P20 AND VR>=1.3 AND VR<=3 AND CLV>=0.65
 AND WIDTH10<=0.15 AND DRY<=0.9 AND TREND AND BIAS20<=0.12;
{此前出现过同定义突破形态，2至10日内回踩后再次转强}
AGE:=BARSLAST(BR);
PB:=COUNT(REF(BR,1),10)>0 AND AGE>=2 AND AGE<=10
 AND L<=M20*1.03 AND C>=M20 AND C>REF(HHV(H,5),1)
 AND DRY<=0.9 AND VR>=1.2 AND CLV>=0.6;
COMMON:=BASE AND R120>=90 AND R250>=85 AND TREND
 AND C/HHV(C,120)>=0.9 AND BIAS20<=0.12;
XG:COMMON AND (BR OR PB);
